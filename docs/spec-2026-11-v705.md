### 1.1 `song.voxLayer`

```jsonc
// 钢琴：右手两个声部、左手一个声部 → 3 行旋律声部，第 1 行是第 0 行的层
"voxVoices":    ["右手", "右手·2", "左手"],
"voxNotations": ["staff", "staff", "bass"],
```

| 条目 | 规则 |
|---|---|
| 含义 | 按旋律声部下标的可选数组。`voxLayer[i]===1`：第 i 行是层，画在第 i−1 行（宿主）的谱表上，作为它的第二声部。`null`：独占一张谱表 |

| 长度 | 必须等于 `voxVoices.length`；不等整组不生效，migrate 载入时连同见证删掉（D-3；长度失配只会来自旧版本增删过行，不可能复原） |
| 位置 | 层行必须紧跟宿主（相对编码的本义）；每张谱表最多 1 层＝最多 2 个声部，第 3 个起导入时计 `skip.voices`（§4.8） |

### 1.2 合法性：`voxLayerLinks` / `voxLayerOk`

两个纯函数，放进 PQ_KIND 共享段（§2.1），编辑器与服务端同一份。

```js
/* 结构链接：数组形状与相对值＋见证＋组边界，不看记谱法（插入点顺延 voxInsAt 用；§3.3） */
function voxLayerLinks(s){
  const a = s && s.voxLayer; if(a===undefined || a===null) return null;          /* 老谱快速返回，不分配数组 */
  const n = s && Array.isArray(s.voxVoices)? s.voxVoices.length : 0;
  if(!Array.isArray(a) || a.length!==n) return null;
  const mx = s.mixer && typeof s.mixer==='object'? s.mixer : null;
  let cut = null;                                            /* cut[i]=1：第 i−1 行与第 i 行之间是某个显式合法组的边界 */
  if(Array.isArray(s.voxGroups)) for(const g of s.voxGroups)
    if(g && (g.type==='brace'||g.type==='bracket') && g.from>=0 && g.to<n && g.to>g.from){ (cut = cut || new Uint8Array(n+1))[g.from] = 1; cut[g.to+1] = 1; }
  let out = null;
  for(let i=1; i<n; i++){ const w = mx && mx['vox:'+i];
    if(a[i]===1 && a[i-1]!==1 && w && typeof w==='object' && w.lay===1 && !(cut && cut[i]))   /* 宿主按原值判，不递归 */
      (out = out || new Array(n).fill(0))[i] = 1; }
  return out;
}
/* 生效（分类 / 视图 / 导出 / 播放 / v70.6 刻版都读它）；mode==='sync'：一致对（谱表级同步用，不要求是五线谱——宿主暂时切成简谱、再切回五线谱要能复原，§1.4） */
function voxLayerOk(s, mode){
  const L = voxLayerLinks(s); if(!L || s.chordOnly || s.func) return null;
  const vn = Array.isArray(s.voxNotations)? s.voxNotations : [];
  const co = i=>{ const v=(Array.isArray(s.voxClefOct)? s.voxClefOct : [])[i]; return v===1 || v===-1? v : 0; };
  const tr = i=>{ const v=(Array.isArray(s.voxTransp)? s.voxTransp : [])[i]; return typeof v==='number' && isFinite(v)? Math.round(v) : 0; };
  let out = null;
  for(let i=1; i<L.length; i++)
    if(L[i] && (mode==='sync' || isStaffNota(vn[i])) && vn[i]===vn[i-1] && co(i)===co(i-1) && tr(i)===tr(i-1)) (out = out || new Array(L.length).fill(0))[i] = 1;
  return out;
}
```

- **组边界**：生效层对不可能跨组边界——v70.5 写组的地方（导入 D-11、accPinGroup :19725、partSong merge :19990、→G :43094）都让组覆盖层；跨了边界的只会来自旧版本整组重写（配伴奏、换谱类、→钢琴伴奏都写 `brace {1,2}`，:7351 / :42595 / :43069 / :41477）。审核探针 `review_compat/probe_old.js` 实测：组边界单独拦得住重建与配伴奏两类，拦不住先删再加；见证四类全拦；两条取「与」。
- `cut[g.from]` 用 `Uint8Array`：`from` 是非整数等坏值时下标赋值被忽略，不抛错。

**坏数据一律当 null（按独立谱表），不抛错、不改入参、不挂死**（单遍，1e5 行草稿实测 8ms，scratchpad `spec705/probe2.js`）：

| 类别 | 例 | 结果 |
|---|---|---|
| 越界 | `voxLayer[0]===1` | 第 0 行独立；若 `voxLayer[1]` 也是 1，第 1 行因「上一行原值是 1」同样独立 |
| 指向自己 / 绝对下标误写 | `0`、`2`、`-1` | 当 null（只认 1，所以写不出环） |
| 链式 / 成环 | `[null,1,1]` | 只有第 1 行是层，第 2 行独立 |
| 记谱法不同 | 宿主 `staff`、层 `bass`；宿主简谱、层五线 | 当 null（两向都测） |
| 谱号八度 / 移调不同 | 旧版本只改了宿主的 `voxClefOct` / `voxTransp` | 当 null |
| 长度不等 | 长 / 短 | 整组 null |
| 非数组 | 对象、字符串、数字 | 整组 null |
| 值类型 | `"1"`（清洗前）、`true`、`NaN` | 当 null（清洗后 `"1"`→1 生效） |
| 整曲形态 | `chordOnly`、`func` | 整组 null |
| 缺见证 | `voxLayer[i]===1` 但 `mixer['vox:i']` 不存在 / 不是对象 / `lay` 不是数字 1 | 这一行独立（手写或 AI 写的 JSON 漏了见证＝按独立谱表，安全方向） |
| 跨组边界 | 显式合法组 `from===i` 或 `to===i−1` | 这一行独立 |

- **`migrate`（index.html:9617）只删长度失配**（D-3）：`voxLayer` 不是数组或长度≠`voxVoices.length` → 删 `voxLayer` 与全部 `mixer['vox:i'].lay`（与 :9629 同一处、同一口径）；其余不生效的值（记谱法 / 八度 / 移调不一致、缺见证）原样留在数据里——宿主切简谱再切回五线谱时层自动恢复。
- 服务端 `viewOf`（api/kindname.js:498）不跑 migrate，所以校验必须在 PQ_KIND 里自带（§2.6）。

### 2.2 视图形态

```js
const VIEW = new WeakMap();   /* 视图 → {src, rows}；每次新建视图，不会过期 */
function collapseLayers(s, mode){
  const R = layerRows(s, mode); if(!R) return s;
  const pick = a=> Array.isArray(a)? R.rows.map(rs=>a[rs[0]]) : a;
  const V = Object.assign({}, s, {voxVoices:pick(s.voxVoices), voxNotations:pick(s.voxNotations), voxDual:pick(s.voxDual),
                                   voxClefOct:pick(s.voxClefOct), voxTransp:pick(s.voxTransp), voxShort:pick(s.voxShort)});
  delete V.voxLayer;                                        /* 嵌套调用恒等：视图上再 collapse 返回它自己 */
  if(Array.isArray(s.voxGroups)) V.voxGroups = s.voxGroups
    .map(g=> g && typeof g==='object'? Object.assign({}, g, {from:R.of[g.from], to:R.of[g.to]}) : g)
    .filter(g=> !(g && typeof g==='object') || !(g.from!=null && g.to!=null) || g.to>g.from);   /* 映射后 to≤from 的丢掉；坏组原样交给 groupsOf 过滤 */
  VIEW.set(V, {src:s, rows:R.rows});
  return V;
}
```

### 2.3 分类视图 `'kind'`：器乐层折叠，声乐层照独立声部算

```js
function layerRows(s, mode){
  const ok = voxLayerOk(s); if(!ok) return null;
  let fold = ok;                                            /* fold[i]=1：第 i 行折进上一行 */
  if(mode==='kind'){
    const P = collapseLayers(s), pr = layerRows(s), gs = groupsOf(P);   /* 物理视图上的组（单元下标） */
    fold = ok.slice();
    pr.rows.forEach((rs, k)=>{ if(rs.length<2) return;
      const inBrace = gs.some(g=>g.type==='brace' && k>=g.from && k<=g.to);
      if(!inBrace && voxHasLyrics(s, rs[0], rs[1])) fold[rs[1]] = 0; });   /* 声乐层：不折 */
    if(!fold.some(Boolean)) return null;
  }
  const rows=[], of=[];
  for(let i=0; i<ok.length; i++){ if(fold[i]) rows[rows.length-1].push(i); else rows.push([i]); of[i]=rows.length-1; }
  return {rows, of};
}
```

- **声乐层**＝所在物理谱表有真歌词（宿主或层任一行），且这张谱表不在花括号组里。声乐层在分类上当独立声部——等价于剥掉 `voxLayer` 后 v70.4 对同一份 JSON 的叫法。
- **器乐层**＝其余（钢琴谱表、无词的小提琴双声部、无词旋律）。器乐层在分类上消失——等价于删掉层行。

### 3.2 `voxMetaV` 的 voxLayer 分支

调用约定同现行：调用时 `voxVoices` 已增删完；`n0=n−ins+del` 是增删前的行数。

```js
function voxLayerSplice(s, at, del, ins){                  /* 在 voxMetaV 末尾调用 */
  const a = s.voxLayer; if(a===undefined) return;
  const n = (s.voxVoices||[]).length, n0 = n - ins + del;
  if(!Array.isArray(a) || a.length!==n0){ delete s.voxLayer; return; }   /* 本就失配（migrate 已删过，这里是兜底）：改声部时收掉 */
  a.splice(at, del, ...Array.from({length:ins}, ()=>null));
  const j = at + ins;
  if((del>0 || ins>0) && j<n && a[j]===1) a[j] = null;    /* 改动块之后紧邻的那一行换了前邻：删宿主 → 层转独立；插在宿主与层之间 → 层转独立（兜底） */
  if(a.every(x=>x==null)) delete s.voxLayer;
}
/* 见证清扫：在 mixer 改键之后调（voxMetaSplice 末尾；追加分支在 voxMetaV 之后）。voxMetaV 内部不调——那时 mixer 还是旧下标，
   会把还没挪过去的见证删掉（例：[X, A, A·2] 删 0，voxLayer 已是 [null,1]，mixer 还是 vox:2） */
function voxLayWSync(s){ if(!s.mixer) return; const a = Array.isArray(s.voxLayer)? s.voxLayer : null;
  Object.keys(s.mixer).forEach(k=>{ const m=/^vox:(\d+)$/.exec(k), e=s.mixer[k]; if(m && e && typeof e==='object' && 'lay' in e && !(a && a[+m[1]]===1)) delete e.lay; }); }
/* 整组重写行之前用（配伴奏、→钢琴伴奏、restruct、换谱类兜底）：删 voxLayer 与全部见证 */
function voxLayerDrop(s){ delete s.voxLayer; voxLayWSync(s); }
```

| 语义 | 例（调用前 → 调用后） |
|---|---|
| 层之后插行 | `[A, A·2, B]`＋在 2 插 → `[null,1,null,null]`，层仍挂 A |
| 宿主与层之间硬插（调用方没顺延） | `[A, A·2, B]`＋在 1 插 → 层转独立，全 null 删键 |
| 删宿主 | `[P, A, A·2]` 删 1 → 层转独立（不会挂到 P 上） |
| 删首行宿主 | `[A, A·2, B]` 删 0 → 删键 |
| 删层 | `[A, A·2, B, B·2]` 删 1 → `[null,null,1]`，别的谱表不受影响 |
| 删宿主前面的行 | `[X, A, A·2]` 删 0 → `[null,1]`，相对值不用改 |
| 末尾追加（voxMetaV 路径） | `[A, A·2]`＋追加 → `[null,1,null]` |
| 原本失配 | `[A, A·2, B]` 但 voxLayer 长 2 → 改声部时删键 |
| 坏数据链 | `[A, A·2, A·3]`＝`[null,1,1]` 删 1 → 第 2 行不会复活成合法层 |

### 4.2 声部预扫与归属（R1–R4）

在每个**源 part**（splitStaves 删 `<staff>` 之前）扫一遍全部小节，游标规则同解析（backup 回退、forward 前进、非和音非倚音的 `<note>` 推进）。

- **R1 声部身份**：`<voice>` 文本是声部 id；缺 `<voice>` 的音记作 `''`（排在所有数字之前，等于「宿主」，同现状永不跳过）；`<chord/>` 缺 voice 时继承主音；倚音按自己的 voice 归行（缺省同下一个主音）。
- **R2 编号约定与归属谱表**（D-21，按**源 part 整体**判，不逐个声部判）：
  1. 约定判定：对每个声部 v，取它在各谱表上的**非和音、非倚音**的 `<note>`（**含休止**）；只要有一个 v 在**同一小节里**、两张不同谱表上的时间区间有重叠，就判这个 part「按谱表编号」（两谱表各自从 1 编号，如 `{1,2@1; 1,2@2}`）。和音（`<chord/>`）不参与重叠判定——和弦里分到另一谱表的音与主音同时刻，是跨谱表和弦，不是另一个声部。
  2. 按谱表编号的 part：**全部声部**一律用 `(v, st)` 作键（`st`＝这个 `<note>` / `<forward>` 自己的 `<staff>`，缺省 1；和音、倚音同样取自己的 staff）。这样两个 voice 2 即使从不在同一小节出现，左手的 voice 2 也不会被当成跨谱表音搬进右手。
  3. 其余 part：键＝v；`home(v)`＝v 的非和音非倚音音（含休止）在各谱表的个数取多数，平票取小号；v 落在别的谱表上的音是**跨谱表音**，搬回 home 行（R12）。`<forward>` 带 `<voice>` 时键＝它的 voice（落 home 行），不带 voice 的 forward 只推游标。
  ——这样不会把左手并进右手（反例夹具 F11、F17，§10.2）。
- **R3 每谱表排名，全曲固定**：home 在本谱表、且在本谱表至少有一个**有音高**音的声部，按 voice 号数值升序：rank0＝宿主，rank1＝层（紧跟宿主，`voxLayer=1`），rank≥2 丢弃并计 `skip.voices`（主音、和音、倚音都计，休止不计，`vBy.rank3`）。纯休止声部不排名、不建层、不计数；本谱表一个有音高的声部都没有时，取最小的纯休止声部当宿主（保住整小节休止）。**例外：自家文件**（有 `pyq-voice-names` 且它点名的行数与按 voice 号排出来的行数对得上）按名单建层，层全是休止 / 只有 forward 也建（否则自家往返时空层整行消失、voxLayer 与行数对不上，§15 I-8）。
- **R4 两种编号都认**：自家旧导出 `{1@1, 2@2}`、MuseScore / OpenScore `{1,2@1; 5,6@2}`、本版自家分层导出同 MuseScore（§7.3）。左手仍按 `<staff>` 判（splitStaves k=2），与 voice 号无关。曲库 10 首全是 `{1@1, 5@2}`（§9）。
- **R11 声部交叉**：不按音高重排，宿主 / 层只由 voice 号决定。

预扫产物（每源 part 一份）：`keyOf(note)`（voice 键，可能是 `(v, st)`）、`hostOf[k]`、`layerOf[k]`、`dropped`（rank≥2 的键集）、`xs`（跨谱表音数，进 `lossy.xstaff`）。

### 4.3 层行生成

1. **splitStaves**（:45532–45557）：克隆第 k 谱表时，删掉 `keyOf` 的谱表 ≠ k 的音（非和音非倚音换成等长 `<forward data-pyq-x="1">`，和音 / 倚音直接删）；`keyOf` 的谱表＝k 但 `<staff>`≠k 的跨谱表音**留下**。`<direction>` / `<harmony>` 仍按 `<staff>` 分（:45544–45547）。`data-pyq-cur`（:45543 / :45546）现已无人读，保留不动；新补的 forward 只打 `data-pyq-x`。
2. **层克隆**（插在 :45558–45561 的 partsAll 组装之后）：对每个五线谱虚拟 part vp，若它的谱表有 rank1 声部，且不属于 R5 的情形，`vp.cloneNode(true)` 一份层 vp，紧跟宿主放进序列，打 `data-pyq-lay="1"`、`data-pyq-vh="<宿主键>"`、`data-pyq-vl="<层键>"`。**方向记号按键分流（D-9）**：层 vp 里只留 `<voice>`＝层键的 `<direction>`（不带 voice 的、voice＝宿主键的、voice 是被丢声部的一律删），并删全部 `<harmony>`、`<sound>`；宿主 vp 里只删 `<voice>`＝层键的 `<direction>`。这样 Finale / Dolet 一类在每个 direction 上都写 `<voice>` 的文件，力度、楔形、踏板只挂一次、播放不叠加。
3. **R5 不建层**（第二声部计 `skip.voices`）：宿主是简谱（score-part `<group>jp`，:45766–45767；`vBy.jp`）、六线谱 part（`vBy.tab`）、宿主被 part 上限截掉（只计一次 `skip.parts`，层音不再计）、层超出总轨上限（`vBy.cap`，§4.5）。
4. **克隆自动得到的**：层 vp 读同一个首个 `<clef>`（:45758）→ voxNotations / voxClefOct（:45762–45763）相同；同一个 `<transpose>`（:45610 的 csh）→ voxTransp（:45764）相同；同一个 score-part `<group>`（:45766）→ voxDual / pianoAcc 相同；同一个 part id → 钢琴音色 mixer（:45688）相同。规格断言这几项逐项相等（§10.3 tv_v705layer I 段）。

### 4.4 解析（每行一个 vp，主体沿用现有代码）

**R7 时间轴**：每小节 `tcur` 从 0 起；所有非和音、非倚音的 `<note>`（包括别的声部、补出的 forward 替身）都推进 tcur；backup 回退、forward 前进（:45918 不改）。每行另记 `tFill`（本行已写到的时间）；**`tMax` 按谱表算**＝本谱表（宿主行与层行）本小节写到的最大时间，不跨谱表、不按 part 共用（D-19）。反例：自家 v70.4 导出的弱起小节，右手 1 拍、左手 4 个四分休止同在一个 part（审核探针 `review_interop/pk.js` 实测 beatErrors 0）；按 part 共用 tMax 会给右手补 3 拍 hid，v70.5 画成可见休止，弱起也被撑满。

**voxGap 拆法（D-18，按拍位；v70.6「新建第二声部」共用）**：输入是一段空拍 `[t0, t1)` 与本小节拍号（含曲中变拍）。
1. **拍**：x/2、x/4 与 2/8、4/8 一类单拍子，一拍＝`1/分母`；6/8、9/8、12/8、6/16 一类复拍子（分子是 3 的倍数且 >3）与 3/8，一拍＝附点（3 个分母单位）。
2. 空拍正好是整小节 → 按现状写成整小节的一组列（v70.5 由 emptyMea :15880 合成整小节休止画出）。
3. 否则先在每个拍界切开。拍内的零头：单拍子按「起点对齐的最大二分时值」切（从半拍起的半拍＝一个八分）；复拍子一拍内三个单位，后两个可以合成一个四分、前两个不合（Gould 复拍子休止写法）；附点休止只用于复拍子里从拍首起、正好占满一拍的空拍。
4. 再把连续的整拍合并，**只合并成从强位起的时值**：单拍子里两拍合成二分只在偶数拍位起（4/4 第 1、3 拍；3/4 第 1 拍；2/2 一拍本就是二分），4/4 前三拍或后三拍不合成附点二分；复拍子里两拍合成附点二分只在 12/8 第 1、3 拍起；6/8、9/8 不合并整拍以外的组合。
5. 至多 16 列；超出或凑不成整时值的余量 ≥ div/16 才计「空拍」。
- 期望（夹具 F01b–F01d；时间从小节起点按四分计）：4/4 从 1 起空 3 拍 → 四分＋二分（现行从大到小拆成附点二分）；4/4 从 0.5 起到小节末（3.5 拍）→ 八分＋四分＋二分（现行拆成二分＋附点四分，二分休止落在第 1 拍后半）；6/8 从第 2 个八分起到小节末（2.5 拍）→ 四分＋附点四分（现行拆成二分＋八分，跨拍）。审核探针 `review_notation/p1.js` 复现了现行拆法。
- `tabGap`（:46103）不改（D-14），六线谱 forward 空拍照补普通休止、拆法照旧。
